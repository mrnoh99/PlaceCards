```swift
@MainActor
class LanguageManager: ObservableObject {
```

## 2. PlaceCard 모델 다언어화

### 2.1 데이터 구조

```swift
@Model
final class PlaceCard {
    @Attribute(.unique) var id: String
    
    // ★ 다언어 상호명 저장
    var nameByLanguage: [String: String] = [:]
    
    // ★ 다언어 주소 저장
    var addressByLanguage: [String: String] = [:]
    
    // 다른 필드들...
    var coordinates: Coordinates?
    var rating: Double?
    var reviewCount: Int?
    var phone: String?
    var website: String?
    var hoursDetail: [String: String]?
    var amenities: [String]
    var tags: [String]
    var media: MediaBundle
    var createdAt: Date
    var updatedAt: Date
    
    // MARK: - 현재 선택된 언어로 표시될 값
    @Transient
    var currentLanguage: String = "en"
    
    var displayName: String {
        nameByLanguage[currentLanguage] ?? 
        nameByLanguage["en"] ?? 
        nameByLanguage.values.first ?? 
        ""
    }
    
    var displayAddress: String {
        addressByLanguage[currentLanguage] ?? 
        addressByLanguage["en"] ?? 
        addressByLanguage.values.first ?? 
        ""
    }
    
    // 하위 호환성
    var name: String {
        get { displayName }
        set { nameByLanguage["en"] = newValue }
    }
    
    var address: String {
        get { displayAddress }
        set { addressByLanguage["en"] = newValue }
    }
}
```

### 2.2 다언어 PlaceCard 생성

```swift
class PlaceCardService {
    func searchAndCreatePlaceCard(
        query: String,
        languages: [String] = ["ko", "en"]
    ) async throws -> PlaceCard {
        var placeCard = PlaceCard(id: UUID().uuidString)
        
        // 각 언어로 Google API 호출
        for language in languages {
            do {
                let results = try await googlePlacesService.search(
                    query: query,
                    languageCode: language
                )
                
                if let result = results.first {
                    // 언어별 상호명 저장
                    placeCard.nameByLanguage[language] = result.displayName
                    
                    // 언어별 주소 저장
                    placeCard.addressByLanguage[language] = result.address
                    
                    // 언어 상관없이 한 번만 저장 (좌표, 평점 등)
                    if language == "ko" || placeCard.coordinates == nil {
                        placeCard.coordinates = result.coordinates
                        placeCard.rating = result.rating
                        placeCard.reviewCount = result.reviewCount
                        placeCard.phone = result.phone
                        placeCard.website = result.website
                        placeCard.hoursDetail = result.hoursDetail
                    }
                }
            } catch {
                print("Failed to fetch for language \(language): \(error)")
                continue
            }
        }
        
        // 기본값 설정
        if placeCard.nameByLanguage.isEmpty {
            throw PlaceCardError.noResults
        }
        
        return placeCard
    }
}
```

## 3. Google Places API 다언어 호출

### 3.1 Service 구현

```swift
class GooglePlacesService {
    let apiKey: String
    
    func search(
        query: String,
        languageCode: String = "en"
    ) async throws -> [PlaceSearchResult] {
        let url = URL(string: "https://places.googleapis.com/v1/places:searchText")!
        var request = URLRequest(url: url)
        request.httpMethod = "POST"
        request.setValue("application/json", forHTTPHeaderField: "Content-Type")
        request.setValue(apiKey, forHTTPHeaderField: "X-Goog-Api-Key")
        
        let body: [String: Any] = [
            "textQuery": query,
            "languageCode": languageCode  // ← 핵심: 언어 파라미터
        ]
        
        request.httpBody = try JSONSerialization.data(withJSONObject: body)
        
        let (data, response) = try await URLSession.shared.data(for: request)
        
        guard let httpResponse = response as? HTTPURLResponse,
              httpResponse.statusCode == 200 else {
            throw NetworkError.httpError
        }
        
        let decoded = try JSONDecoder().decode(GooglePlacesResponse.self, from: data)
        
        return decoded.places.map { place in
            PlaceSearchResult(
                id: place.id,
                displayName: place.displayName.text,
                address: place.formattedAddress,
                coordinates: (
                    latitude: place.location.latitude,
                    longitude: place.location.longitude
                ),
                rating: place.rating,
                reviewCount: place.userRatingCount,
                phone: place.nationalPhoneNumber,
                website: place.websiteUri,
                hoursDetail: place.regularOpeningHours?.weekdayDescriptions ?? [:]
            )
        }
    }
}

// API 응답 모델
struct GooglePlacesResponse: Decodable {
    let places: [GooglePlace]
}

struct GooglePlace: Decodable {
    let id: String
    let displayName: DisplayName
    let formattedAddress: String
    let location: Location
    let rating: Double?
    let userRatingCount: Int?
    let nationalPhoneNumber: String?
    let websiteUri: String?
    let regularOpeningHours: OpeningHours?
}

struct DisplayName: Decodable {
    let text: String
    let languageCode: String
}

struct Location: Decodable {
    let latitude: Double
    let longitude: Double
}

struct OpeningHours: Decodable {
    let weekdayDescriptions: [String]?
}
```

## 4. Naver Map API 다언어 호출 (백엔드 프록시)

### 4.1 백엔드 (Node.js)

```javascript
// backend/routes/naver.js
const express = require('express');
const axios = require('axios');
const router = express.Router();

// POST /api/naver/search
router.post('/search', async (req, res) => {
    const { query, language } = req.body;
    
    try {
        const naverResponse = await axios.get(
            'https://openapi.naver.com/v1/search/local.json',
            {
                headers: {
                    'X-Naver-Client-Id': process.env.NAVER_CLIENT_ID,
                    'X-Naver-Client-Secret': process.env.NAVER_CLIENT_SECRET
                },
                params: {
                    query: query,
                    display: 10,
                    language: language || 'ko'
                }
            }
        );
        
        res.json({
            language: language,
            success: true,
            items: naverResponse.data.items
        });
    } catch (error) {
        res.status(error.response?.status || 500).json({
            success: false,
            error: error.message
        });
    }
});

module.exports = router;
```

### 4.2 iOS 클라이언트

```swift
class NaverProxyService {
    let proxyBaseURL: URL
    
    func search(
        query: String,
        language: String = "ko"
    ) async throws -> [NaverSearchResult] {
        let url = proxyBaseURL.appendingPathComponent("api/naver/search")
        
        var request = URLRequest(url: url)
        request.httpMethod = "POST"
        request.setValue("application/json", forHTTPHeaderField: "Content-Type")
        
        let body: [String: Any] = [
            "query": query,
            "language": language
        ]
        
        request.httpBody = try JSONSerialization.data(withJSONObject: body)
        
        let (data, response) = try await URLSession.shared.data(for: request)
        
        guard let httpResponse = response as? HTTPURLResponse,
              httpResponse.statusCode == 200 else {
            throw NetworkError.httpError
        }
        
        let decoded = try JSONDecoder().decode(NaverResponse.self, from: data)
        
        return decoded.items.map { item in
            NaverSearchResult(
                id: "\(item.mapx),\(item.mapy)",
                title: item.title,
                address: item.roadAddress ?? item.address,
                coordinates: (
                    latitude: Double(item.mapy) ?? 0,
                    longitude: Double(item.mapx) ?? 0
                )
            )
        }
    }
}
```

## 5. AI (Claude) 다언어 이미지 분석

### 5.1 한국 스크린샷 분석

```swift
class ClaudeVisionService: AIProvider {
    let apiKey: String
    
    func analyzeImageKorean(_ image: UIImage) async throws -> PlaceAnalysisResult {
        guard let imageData = image.jpegData(compressionQuality: 0.8) else {
            throw AnalysisError.invalidImage
        }
        
        let base64Image = imageData.base64EncodedString()
        
        // 한국어 프롬프트
        let koreanPrompt = """
        이 이미지에서 카페 또는 음식점의 정보를 추출해주세요.
        한글로 다음 정보를 JSON 형식으로 반환하세요:
        {
          "placeName": "상호명",
          "address": "주소",
          "phone": "전화번호",
          "hours": "영업시간",
          "confidence": 0.95
        }
        """
        
        return try await analyzeImage(
            base64Image: base64Image,
            prompt: koreanPrompt
        )
    }
    
    func analyzeImageEnglish(_ image: UIImage) async throws -> PlaceAnalysisResult {
        guard let imageData = image.jpegData(compressionQuality: 0.8) else {
            throw AnalysisError.invalidImage
        }
        
        let base64Image = imageData.base64EncodedString()
        
        // 영문 프롬프트
        let englishPrompt = """
        Extract information about the cafe or restaurant from this image.
        Return the following information in JSON format in English:
        {
          "placeName": "Business name",
          "address": "Address",
          "phone": "Phone number",
          "hours": "Operating hours",
          "confidence": 0.95
        }
        """
        
        return try await analyzeImage(
            base64Image: base64Image,
            prompt: englishPrompt
        )
    }
    
    private func analyzeImage(
        base64Image: String,
        prompt: String
    ) async throws -> PlaceAnalysisResult {
        let url = URL(string: "https://api.anthropic.com/v1/messages")!
        var request = URLRequest(url: url)
        request.httpMethod = "POST"
        request.setValue("application/json", forHTTPHeaderField: "Content-Type")
        request.setValue(apiKey, forHTTPHeaderField: "x-api-key")
        request.setValue("2023-06-01", forHTTPHeaderField: "anthropic-version")
        
        let payload: [String: Any] = [
            "model": "claude-3-5-sonnet-20241022",
            "max_tokens": 1024,
            "messages": [
                [
                    "role": "user",
                    "content": [
                        [
                            "type": "image",
                            "source": [
                                "type": "base64",
                                "media_type": "image/jpeg",
                                "data": base64Image
                            ]
                        ],
                        [
                            "type": "text",
                            "text": prompt
                        ]
                    ]
                ]
            ]
        ]
        
        request.httpBody = try JSONSerialization.data(withJSONObject: payload)
        
        let (data, _) = try await URLSession.shared.data(for: request)
        
        // JSON 파싱 및 변환
        let response = try JSONDecoder().decode(ClaudeResponse.self, from: data)
        
        if let textContent = response.content.first(where: { $0.type == "text" }) {
            // JSON 추출 및 파싱
            if let jsonData = textContent.text.data(using: .utf8),
               let analysisResult = try? JSONDecoder().decode(
                   PlaceAnalysisResult.self,
                   from: jsonData
               ) {
                return analysisResult
            }
        }
        
        throw AnalysisError.parsingFailed
    }
}
```

## 6. 통합 흐름

```swift
@MainActor
class PlaceCardCreationViewModel: ObservableObject {
    @Published var selectedImage: UIImage?
    @Published var currentLanguage: String = "en"
    @Published var isLoading: Bool = false
    @Published var placeCard: PlaceCard?
    @Published var error: Error?
    
    private let googleService: GooglePlacesService
    private let naverService: NaverProxyService
    private let claudeService: ClaudeVisionService
    
    // 사용자가 사진 선택
    func handleImageSelected(_ image: UIImage) async {
        isLoading = true
        defer { isLoading = false }
        
        do {
            // 1. AI로 위치명 추출 (한국어)
            let analysisResult = try await claudeService.analyzeImageKorean(image)
            let placeName = analysisResult.placeName
            
            // 2. Google + Naver에서 다언어로 조회
            let placeCard = try await searchPlaceInMultipleLanguages(
                query: placeName,
                languages: ["ko", "en"]
            )
            
            self.placeCard = placeCard
            
        } catch {
            self.error = error
        }
    }
    
    private func searchPlaceInMultipleLanguages(
        query: String,
        languages: [String]
    ) async throws -> PlaceCard {
        var placeCard = PlaceCard(id: UUID().uuidString)
        
        // 각 언어로 동시에 요청
        let results = try await withThrowingTaskGroup(of: (String, PlaceSearchResult).self) { group in
            for language in languages {
                group.addTask {
                    let results = try await self.googleService.search(
                        query: query,
                        languageCode: language
                    )
                    guard let first = results.first else {
                        throw PlaceCardError.noResults
                    }
                    return (language, first)
                }
            }
            
            var results: [(String, PlaceSearchResult)] = []
            for try await result in group {
                results.append(result)
            }
            return results
        }
        
        // 결과를 PlaceCard에 저장
        for (language, result) in results {
            placeCard.nameByLanguage[language] = result.displayName
            placeCard.addressByLanguage[language] = result.address
        }
        
        // 공통 정보 (한 번만 저장)
        if let firstResult = results.first?.1 {
            placeCard.coordinates = firstResult.coordinates
            placeCard.rating = firstResult.rating
            placeCard.reviewCount = firstResult.reviewCount
            placeCard.phone = firstResult.phone
            placeCard.website = firstResult.website
        }
        
        return placeCard
    }
}
```
