## 2. 데이터 흐름

### 2.1 사진 추가 → PlaceCard 생성 플로우

```
[사용자가 사진 선택]
(지도 스크린샷, 인스타그램, 또는 직접 촬영)
         │
         ↓
[AI API (사용자 키)로 이미지 분석]
(위치명, 주소, 설명 추출)
         │
         ↓
[추출된 위치명 후보 확인]
(사용자 수동 확인)
         │
         ↓
[Naver API로 검색]
         │
     ┌───┴───┐
     │       │
   YES      NO
     │       │
     ↓       ↓
[기본정보  [Google API
 획득]     직접 검색]
     │       │
     └───┬───┘
         │
         ↓
[Google API로 상세정보 보강]
(평점, 리뷰, 영업시간, 편의시설)
         │
         ↓
[PlaceCard 생성 & 저장]
(SwiftData)
         │
         ↓
[UI 업데이트]
(갤러리, 지도 새로고침)
```

### 3.5 KeychainService

```swift
class KeychainService {
    static func save(
        key: String,
        value: String,
        service: String = "com.placecards.app"
    ) throws {
        let data = value.data(using: .utf8)!
        
        let query: [String: Any] = [
            kSecClass as String: kSecClassGenericPassword,
            kSecService as String: service,
            kSecAccount as String: key,
            kSecValue as String: data,
            kSecAttrAccessible as String: kSecAttrAccessibleWhenUnlockedThisDeviceOnly
        ]
        
        SecItemDelete(query as CFDictionary)
        
        let status = SecItemAdd(query as CFDictionary, nil)
        guard status == errSecSuccess else {
            throw KeychainError.saveFailed(status: status)
        }
    }
    
    static func load(
        key: String,
        service: String = "com.placecards.app"
    ) throws -> String? {
        let query: [String: Any] = [
            kSecClass as String: kSecClassGenericPassword,
            kSecService as String: service,
            kSecAccount as String: key,
            kSecReturnData as String: true
        ]
        
        var result: AnyObject?
        let status = SecItemCopyMatching(query as CFDictionary, &result)
        
        guard status == errSecSuccess,
              let data = result as? Data,
              let value = String(data: data, encoding: .utf8) else {
            if status == errSecItemNotFound {
                return nil
            }
            throw KeychainError.loadFailed(status: status)
        }
        
        return value
    }
}
```

## 4. ViewModel 계층

### 4.1 PlaceCardViewModel

```swift
@MainActor
class PlaceCardViewModel: ObservableObject {
    @Published var isLoading: Bool = false
    @Published var error: Error? = nil
    @Published var placeCard: PlaceCard? = nil
    
    @Published var selectedImage: UIImage? = nil
    @Published var extractedPlaceName: String? = nil
    @Published var candidates: [String] = []
    
    private let aiProvider: AIProvider
    private let googleService: GooglePlacesService
    private let naverService: NaverProxyService
    private let storageService: StorageService
    
    // MARK: - Image Analysis
    func analyzeImage(_ image: UIImage) async {
        isLoading = true
        defer { isLoading = false }
        
        do {
            // 1. AI로 분석
            guard let imageData = image.jpegData(compressionQuality: 0.8) else {
                throw AnalysisError.invalidImage
            }
            
            let result = try await aiProvider.analyzeImage(
                imageData: imageData,
                prompt: "지도 또는 사진에서 장소명을 추출하세요"
            )
            
            self.extractedPlaceName = result.placeName
            self.candidates = [result.placeName]
            
            // 2. 사용자 확인 후 검증
            
        } catch {
            self.error = error
        }
    }
    
    // MARK: - Search & Create
    func searchAndCreatePlaceCard(placeName: String) async {
        isLoading = true
        defer { isLoading = false }
        
        do {
            // 1. Naver 검색
            let naverResults = try await naverService.search(query: placeName)
            
            if naverResults.isEmpty {
                // 2. Google로 재검색
                let googleResults = try await googleService.search(
                    query: placeName
                )
                guard !googleResults.isEmpty else {
                    throw SearchError.notFound
                }
                
                self.placeCard = try await createPlaceCard(from: googleResults[0])
            } else {
                // 2. Google로 상세정보 보강
                let googleResults = try await googleService.search(
                    query: placeName,
                    location: naverResults[0].coordinates
                )
                
                let baseCard = createBaseCard(from: naverResults[0])
                self.placeCard = try await enrichPlaceCard(
                    baseCard,
                    with: googleResults[0]
                )
            }
            
            // 3. 저장
            if let card = placeCard {
                try storageService.save(card)
            }
            
        } catch {
            self.error = error
        }
    }
}
```

## 7. 에러 처리

```swift
enum PlaceCardsError: LocalizedError {
    // Network
    case networkError(String)
    case apiError(String, statusCode: Int)
    case timeout
    
    // Data
    case validationError(String)
    case decodingError(String)
    case saveFailed(String)
    
    // Security
    case keychainError(String)
    case apiKeyInvalid
    case apiKeyExpired
    
    // User
    case userCancelled
    case noResults
    
    var errorDescription: String? {
        switch self {
        case .networkError(let msg):
            return "네트워크 오류: \(msg)"
        case .apiError(let msg, let code):
            return "API 오류 (\(code)): \(msg)"
        case .timeout:
            return "요청 시간 초과"
        case .apiKeyInvalid:
            return "유효하지 않은 API 키입니다"
        case .noResults:
            return "검색 결과가 없습니다"
        default:
            return "오류 발생"
        }
    }
}
```

## 8. 성능 최적화 전략

### 8.1 이미지 처리
- 저장 전 압축 (1200x1200px max, JPEG 80%)
- Lazy Loading (화면에 보일 때만 로드)
- 썸네일 캐싱

### 8.2 API 호출
- Request Deduplication (동일 요청 중복 제거)
- Response Caching (1시간)
- Rate Limiting (초당 요청 제한)

### 8.3 UI 렌더링
- LazyVStack / LazyVGrid 사용
- Pagination (20개씩 로드)
- Main Thread에서만 UI 업데이트
